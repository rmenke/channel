99 constexpr update_result update() noexcept {
100 update_result result{
101 .move_x = 2 * _error > -_dy, .move_y = 2 * _error < _dx

373inline thick_line_view::iterator thick_line_view::iterator::create(
374 int cur_x, int cur_y, int dx, int dy, int sx, int sy, unsigned int r2
375) {
376 // Determine the axis along which the parallel lines have the
377 // greatest change. This is the major direction: create a
378 // function pointer to access the corresponding component of any
379 // point.
380
381 // clang-format off
382 int &(*const major)(int &, int &) noexcept = dx >= dy ?
383 [](int &x, int &) noexcept -> int & { return x; } :
384 [](int &, int &y) noexcept -> int & { return y; };
385 // clang-format on
386
387 // Generate a perpendicular to the baseline that passes
388 // through the origin. These points will be the offsets from the
389 // baseline to the parallel lines.
390
391 // Find a point that lies on the perpendicular and is sufficiently
392 // far away from the origin.
393
394 int px = -sy, py = +sx;
395
396 // The orientation of the perpendicular is important. The
397 // movement component in the major direction for both the parallel
398 // and perpendicular lines must be the same. If it is not, rotate
399 // the perpendicular an additional 180°.
400
401 if (major(px, py) != major(sx, sy)) {
402 px = +sy, py = -sx;
403 }
404
405 int perp_x = px * dy, perp_y = py * dx;
406
407 // Ensure that the starting point is sufficiently far away. (It
408 // should alredy be except in pathological cases.)
409
410 constexpr static auto length_squared = [](int x, int y) noexcept {
411 return x * x + y * y;
412 };
413
414 while (length_squared(perp_x, perp_y) < r2) {
415 perp_x *= 2, perp_y *= 2;
416 }
417
418 // A predicate that determines if the point is within the radius
419 // from the origin.
420
421 const auto within_radius = [&r2](auto &&pt) {
422 auto &&[x, y] = pt;
423 return length_squared(x, y) <= r2;
424 };
425
426 // Create a view over the perpendicular that only contains the
427 // points within the radius.
428
429 auto perpendicular =
430 thin_line_view(-perp_x, -perp_y, perp_x, perp_y) |
431 std::views::drop_while(std::not_fn(within_radius)) |
432 std::views::take_while(within_radius);
433
434 auto begin = perpendicular.begin();
435 auto end = perpendicular.end();
436
437 int next_x = begin->first;
438 int next_y = begin->second;
439
440 std::vector<thin_line_view::iterator> iterators;
441 error_accumulator parallel_error(dx, dy);
442
443 while (++begin != end) {
444 auto off_x = std::exchange(next_x, begin->first);
445 auto off_y = std::exchange(next_y, begin->second);
446
447 iterators.emplace_back(
448 cur_x + off_x, cur_y + off_y, sx, sy, parallel_error
449 );
450
451 auto const x_will_change = (next_x - off_x) != 0;
452 auto const y_will_change = (next_y - off_y) != 0;
453
454 // If the perpendicular is about to make a diagonal move, then
455 // update the parallel error accumulator.
456
457 if (x_will_change && y_will_change) {
458 auto [move_x, move_y] = parallel_error.update();
459
460 // If the parallel iterator had a diagonal move as well
461 // then this will leave a pixel-wide gap between the
462 // parallel lines. Murphy solved this by drawing an
463 // additional parallel, effectively turning a single
464 // diagonal move into two square moves.
465
466 if (move_x && move_y) {
467 major(off_x, off_y) += major(px, py);
468 iterators.emplace_back(
469 cur_x + off_x, cur_y + off_y, sx, sy, parallel_error
470 );
471 }
472 }
473 }
474
475 iterators.emplace_back(
476 cur_x + next_x, cur_y + next_y, sx, sy, parallel_error
477 );
478
479 return iterator{std::move(iterators)};
480}
