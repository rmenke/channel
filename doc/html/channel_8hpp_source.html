443inline thick_line_view::iterator thick_line_view::iterator::create(
444 point const &current, point const &delta, point const &step,
445 unsigned int r2
446) {
447 auto &&[dx, dy] = delta;
448 auto &&[sx, sy] = step;
449
450 // This gets complicated...
451 //
452 // Determine the axis along which the parallel lines have the
453 // greatest change. Make some pointers to member fields for later
454 // use.
455
456 int point::*const major = delta.x >= delta.y ? &point::x : &point::y;
457 int point::*const minor = delta.x >= delta.y ? &point::y : &point::x;
458
459 // Generate a perpendicular to the baseline that passes
460 // through the origin. These points will be the offsets from the
461 // baseline to the parallel lines.
462
463 // Find a point that lies on the perpendicular and is sufficiently
464 // far away from the origin.
465
466 point perp_step = {-sy, +sx};
467
468 // The orientation of the perpendicular is important. The
469 // movement component in the major direction for both the parallel
470 // and perpendicular lines must be the same. If it is not, rotate
471 // the perpendicular an additional 180°.
472
473 if (perp_step.*major != step.*major) {
474 perp_step *= -1;
475 }
476
477 point perp_endpt = {
478 perp_step.x * dy,
479 perp_step.y * dx,
480 };
481
482 // Ensure that the starting point is sufficiently far away. (It
483 // should alredy be except in pathological cases.)
484
485 while (perp_endpt.length_squared() < r2) {
486 perp_endpt *= 2;
487 }
488
489 // A predicate that determines if the point is within the radius
490 // from the origin.
491
492 const auto within_radius = [&r2](auto &&pt) {
493 return pt.length_squared() <= r2;
494 };
495
496 // Create a view over the perpendicular that only contains the
497 // points within the radius.
498
499 auto perpendicular =
500 thin_line_view(-perp_endpt, perp_endpt) |
501 std::views::drop_while(std::not_fn(within_radius)) |
502 std::views::take_while(within_radius);
503
504 auto begin = perpendicular.begin();
505 auto end = perpendicular.end();
506 auto next = *begin;
507
508 std::vector<thin_line_view::iterator> iterators;
509 error_accumulator parallel_error(delta);
510
511 while (++begin != end) {
512 auto offset = std::exchange(next, *begin);
513
514 iterators.emplace_back(current + offset, parallel_error, step);
515
516 auto const x_will_change = (next.x - offset.x) != 0;
517 auto const y_will_change = (next.y - offset.y) != 0;
518
519 // If the perpendicular is about to make a diagonal move, then
520 // update the parallel error accumulator.
521
522 if (x_will_change && y_will_change) {
523 auto [move_x, move_y] = parallel_error.update();
524
525 // If the parallel iterator had a diagonal move as well
526 // then this will leave a pixel-wide gap between the
527 // parallel lines. Murphy solved this by drawing an
528 // additional parallel, effectively turning a single
529 // diagonal move into two square moves.
530
531 if (move_x && move_y) {
532 offset.*major += perp_step.*major;
533 iterators.emplace_back(
534 current + offset, parallel_error, step
535 );
536 offset.*minor += perp_step.*minor;
537 }
538 }
539 }
540
541 iterators.emplace_back(current + next, parallel_error, step);
542
543 return iterator{std::move(iterators)};
544}
